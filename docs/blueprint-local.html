§02สถาปัตยกรรม

โปรเซสเดียว ฟังที่ 127.0.0.1 เท่านั้น เบราว์เซอร์คือหน้าจอ ดิสก์คือที่เก็บของ ไม่มีอะไรวิ่งออกเน็ตนอกจากการเรียก API ของ Google และ Anthropic ด้วย key ของผู้ใช้เอง

ข้อบังคับ ห้ามพลาด

เซิร์ฟเวอร์ต้อง bind 127.0.0.1 เท่านั้น ห้าม 0.0.0.0 ถ้า bind ทุก interface แล้วผู้ใช้ไปนั่งร้านกาแฟ คนอื่นในวงเดียวกันเปิดหน้าเว็บนี้ได้ และใช้ API key ของเขายิง TTS ได้ฟรี ๆ — ค่านี้ต้อง hardcode ไม่ใช่ config

§03Tech stack และเหตุผล

เกณฑ์ตัดสินเปลี่ยนไปจากเวอร์ชัน SaaS อย่างสิ้นเชิง — เดิมคือ "สเกลได้ไหม" ตอนนี้คือ "คนที่ไม่ใช่โปรแกรมเมอร์ติดตั้งผ่านไหม" ทุกบรรทัดข้างล่างเลือกด้วยเกณฑ์นี้

ชั้นเลือกใช้เหตุผล
Web servernode:http ล้วน ๆไม่มี dependency → npm install เร็วและแทบไม่มีอะไรพังได้ · การอัปโหลดใช้ PUT ไฟล์ดิบทีละไฟล์ ไม่ต้องมีตัว parse multipart
FrontendVite + React · commit ไฟล์ที่ build แล้วผู้ใช้ปลายทางไม่ต้อง build และไม่ต้องลง dev dependency เลย · คนที่จะแก้ UI ค่อยรัน npm run dev
ฐานข้อมูลnode:sqliteติดมากับ Node 22+ ไม่ต้องติดตั้งอะไร · ไฟล์เดียว ก็อปไปไหนก็ได้ · เขียนผ่าน adapter เผื่อสลับกลับไป Mongo ตอนขึ้น cloud
คิวงานคิวในโปรเซส + สถานะใน SQLiteผู้ใช้คนเดียว รันทีละ 1–2 งาน · Redis เป็นของที่ต้องติดตั้งเพิ่มโดยไม่ได้อะไรกลับมา
ffmpegโหลด binary ให้อัตโนมัติตอนเปิดครั้งแรกนี่คือด่านตายของ Windows user · ถ้าเครื่องมี ffmpeg อยู่แล้วก็ใช้ตัวนั้น
เลน Bhyperframesdependency ตัวเดียวของโปรเจกต์ · มันโหลด Chrome ของตัวเอง ผู้ใช้ไม่ต้องจัดการ
ฟอนต์Kanit ในโฟลเดอร์โปรเจกต์เลน B ฝังเป็น data URI · เลน A ชี้ผ่าน fontsdir — ไม่ต้องติดตั้งฟอนต์ลงเครื่อง
TTS / LLMGemini เป็นหลัก · Claude ถ้ามี keyผู้ใช้ขอ Gemini key ฟรีได้ใน 2 นาที · ถ้าบังคับให้มี Claude key ด้วยจะเสียคนไปครึ่ง
ผลรวมของการเลือกทั้งหมด

node_modules ของผู้ใช้ปลายทางมี hyperframes ตัวเดียว และของนั้นจำเป็นเฉพาะตอนใช้สไตล์เลน B — คนที่ใช้แค่สไตล์ธรรมดารันได้โดยไม่ต้อง npm install อะไรเลย

ตาราง SQLite

-- ดัชนีของโปรเจกต์ เพื่อให้หน้ารายการโหลดเร็วโดยไม่ต้องอ่านทุกโฟลเดอร์
CREATE TABLE projects (
  id           TEXT PRIMARY KEY,      -- = ชื่อโฟลเดอร์
  title        TEXT NOT NULL,
  product_json TEXT NOT NULL,         -- brief ทั้งก้อน
  wizard_step  INTEGER DEFAULT 1,
  created_at   INTEGER, updated_at INTEGER
);

CREATE TABLE renders (
  id           TEXT PRIMARY KEY,
  project_id   TEXT NOT NULL REFERENCES projects(id) ON DELETE CASCADE,
  kind         TEXT NOT NULL,         -- draft | final
  lane         TEXT NOT NULL,         -- ass | hyperframes
  state        TEXT NOT NULL,         -- queued|ingesting|…|ready|failed|canceled
  progress     INTEGER DEFAULT 0,
  timeline_json TEXT,
  outputs_json TEXT,
  error_json   TEXT,
  attempts     INTEGER DEFAULT 0,
  created_at   INTEGER, started_at INTEGER, finished_at INTEGER
);
CREATE INDEX renders_by_project ON renders(project_id, created_at DESC);
CREATE INDEX renders_pending    ON renders(state, created_at);

-- แคชเสียง: คีย์เดียวกับชื่อไฟล์ใน data/cache/tts
CREATE TABLE voice_cache (
  key TEXT PRIMARY KEY,               -- sha256(provider|voice|speed|text)
  duration_ms INTEGER NOT NULL,
  provider TEXT, voice TEXT, created_at INTEGER
);

CREATE TABLE settings (key TEXT PRIMARY KEY, value TEXT);  -- ffmpeg path, เวอร์ชัน, ฯลฯ

ทุกการเข้าถึงข้อมูลผ่าน interface เดียว (listProjects, saveRender, …) โดยมี SqliteStore เป็นตัวเดียวที่ implement ตอนนี้ ตอนขึ้น cloud ค่อยเขียน MongoStore เพิ่ม โดยไม่ต้องแตะโค้ด pipeline เลย — ห้ามเรียก SQL ตรงจากที่อื่นนอกจากไฟล์ store

§07คิวงานในโปรเซส

  • รันพร้อมกันได้ 1 งาน สำหรับเลน B และ 2 งานสำหรับเลน A — ผู้ใช้กำลังใช้เครื่องอยู่ด้วย ห้ามกินเครื่องจนใช้งานอย่างอื่นไม่ได้
  • คิวร่างได้คิวก่อนคิวตัวจริงเสมอ — คนกดร่างแล้วนั่งรออยู่หน้าจอ
  • สถานะเขียนลง SQLite ทุกครั้งที่เปลี่ยน ปิดโปรแกรมแล้วเปิดใหม่ยังเห็นว่างานไหนค้าง
  • ทุกงานทำใน temp dir ของตัวเอง แล้วลบใน finally — ดิสก์เต็มบนเครื่องคนอื่นคือปัญหาที่เราตามไปแก้ไม่ได้
  • จับ SIGINT ผู้ใช้กด Ctrl+C บ่อยกว่าที่คิด — ต้องฆ่า child process ของ ffmpeg/Chrome ให้หมด ไม่งั้นเหลือ process ผีค้างเครื่อง
  • Gemini free tier ยิงขนานไม่ได้ — TTS ยิงทีละคำขอ และอ่าน retryDelay ที่ Google ส่งกลับมา (วัดมาแล้วในสัปดาห์แรก)

§08First-run setup — ส่วนที่สำคัญที่สุด

กลุ่มเป้าหมายหลักคือคนทั่วไปที่ไม่ใช่สายเทค สำหรับพวกเขา ความยากทั้งหมดของโปรเจกต์นี้อยู่ในสิบนาทีแรก ถ้าผ่านตรงนี้ไปได้ ที่เหลือคือกดปุ่มตามหน้าเว็บ

รายละเอียดที่ตัดสินว่าคนจะใช้ได้หรือไม่ได้

เรื่องทำอย่างไร
ffmpegหาตามลำดับ: data/bin/ → PATH ของเครื่อง → ดาวน์โหลด · Windows ดึงจาก gyan.dev, macOS/Linux ใช้ static build · ตรวจว่ามี libass ก่อนบันทึกว่าใช้ได้
Node ไม่มี.bat ตรวจก่อน ถ้าไม่มีให้เปิดหน้าดาวน์โหลด Node พร้อมข้อความภาษาไทย ไม่ใช่ปล่อยให้ขึ้น 'node' is not recognized แล้วหน้าต่างปิดหายไป

เก็บ keyเขียนลง .env ที่อยู่ใน .gitignore แล้ว · ห้ามส่งค่า key กลับไปหน้าเว็บ แสดงแค่ 4 ตัวท้าย
พอร์ตชนกันลอง 4321 → 4322 → 4323 … แล้วเปิดเบราว์เซอร์ไปพอร์ตที่ได้จริง

§13ความเสี่ยง

ความเสี่ยงระดับการรับมือ
ติดตั้งไม่ผ่าน แล้วเงียบหายไปสูงความเสี่ยงอันดับหนึ่งของเวอร์ชันนี้ · ต้องทดสอบบน Windows เครื่องเปล่าจริง ไม่ใช่เครื่อง dev ที่มีของครบ · มีวิดีโอสอน 3 นาที
Gemini free tier ยิงขนานไม่ได้ / โควตาหมดสูงยิงทีละคำขอ + อ่าน retryDelay + แคชทุกท่อน · ขึ้นข้อความบอกตรง ๆ ว่าโควตาหมดและรอได้เมื่อไหร่
เครื่องผู้ใช้ช้าเกินจนเลน B ใช้ไม่ได้กลางวัดเวลาเรนเดอร์ครั้งแรกแล้วจำไว้ · ถ้าเกิน 5 นาทีให้แนะนำสไตล์เลน A พร้อมบอกเหตุผล
ผู้ใช้ทำ .env หลุดขึ้น GitHubกลาง.gitignore ตั้งแต่ commit แรก · README เตือน · setup ไม่แสดง key เต็มที่ไหนเลย
ลิขสิทธิ์คลิปที่ผู้ใช้เอามาใช้กลางเตือนใน README และในหน้าอัปโหลด · ความรับผิดชอบอยู่ที่ผู้ใช้ เพราะเขารันบนเครื่องตัวเอง
รองรับสามระบบปฏิบัติการกลางทุก path ผ่าน node:path · ทดสอบ Windows เป็นหลัก (กลุ่มเป้าหมายส่วนใหญ่)
สิ่งที่หายไปพร้อมการเปลี่ยนทิศ

ความเสี่ยงเดิม 4 ข้อหายไปเลย: Plesk รันงานหนักไม่ได้ · SSE โดน buffer · ดิสก์เซิร์ฟเวอร์เต็ม · ต้นทุน AI บานเพราะผู้ใช้กด regenerate รัว — ทั้งหมดกลายเป็นปัญหาของเครื่องผู้ใช้เอง ซึ่งเขาควบคุมได้และเราไม่ต้องจ่าย
